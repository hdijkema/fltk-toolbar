cmake_minimum_required(VERSION 3.21)
project(fltk-toolbar VERSION 0.1.1 LANGUAGES CXX)
if(APPLE)
    enable_language(OBJCXX)
endif()
option(FLTK_TOOLBAR_BUILD_TESTS "Build toolbar contracts" ${PROJECT_IS_TOP_LEVEL})
option(FLTK_TOOLBAR_BUILD_EXAMPLE "Build the toolbar example" ${PROJECT_IS_TOP_LEVEL})
if(NOT TARGET fltk::fltk)
    find_package(FLTK 1.4.5 CONFIG QUIET)
endif()
if(NOT TARGET fltk::fltk)
    include(FetchContent)
    foreach(part TEST EXAMPLES FLUID FLTK_OPTIONS GL FORMS HTML_DOCS PDF_DOCS)
        set(FLTK_BUILD_${part} OFF CACHE BOOL "" FORCE)
    endforeach()
    FetchContent_Declare(fltk GIT_REPOSITORY https://github.com/fltk/fltk.git GIT_TAG release-1.4.5 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(fltk)
endif()
add_library(fltk-toolbar STATIC src/fltk_toolbar.cpp src/fltk_toolbar_svg.cpp)
add_library(fltk-toolbar::fltk-toolbar ALIAS fltk-toolbar)
target_compile_features(fltk-toolbar PUBLIC cxx_std_17)
target_include_directories(fltk-toolbar PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fltk-toolbar PUBLIC fltk::fltk fltk::images)
if(MSVC)
    target_compile_options(fltk-toolbar PRIVATE /utf-8 /W4)
endif()
if(FLTK_TOOLBAR_BUILD_TESTS)
    enable_testing()
    add_executable(fltk-toolbar-tests tests/contracts.cpp)
    target_link_libraries(fltk-toolbar-tests PRIVATE fltk-toolbar)
    add_test(NAME toolbar-contracts COMMAND fltk-toolbar-tests)
    set_tests_properties(toolbar-contracts PROPERTIES TIMEOUT 30)
endif()
if(FLTK_TOOLBAR_BUILD_EXAMPLE)
    add_executable(fltk-toolbar-example WIN32 MACOSX_BUNDLE examples/toolbar.cpp)
    target_link_libraries(fltk-toolbar-example PRIVATE fltk-toolbar)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS fltk-toolbar EXPORT fltk-toolbar-targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT fltk-toolbar-targets NAMESPACE fltk-toolbar:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fltk-toolbar)
configure_package_config_file(cmake/fltk-toolbar-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/fltk-toolbar-config.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fltk-toolbar)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/fltk-toolbar-config-version.cmake" COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/fltk-toolbar-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/fltk-toolbar-config-version.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fltk-toolbar)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fltk-toolbar)
